```html
<script>
(function(){
  var ov = document.getElementById('introOverlay');
  if (!ov) return;
  var skipBtn = document.getElementById('introSkip');
  var prefersReduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var isNarrow = window.matchMedia('(max-width:900px)').matches;
  var seen = false;
  var done = false;
  try { seen = sessionStorage.getItem('crest_intro_seen') === '1'; } catch (e) {}

  function finish(){
    if (done) return;
    done = true;
    ov.style.display = 'none';
    document.body.classList.remove('intro-lock');
    try { sessionStorage.setItem('crest_intro_seen', '1'); } catch (e) {}
    document.removeEventListener('keydown', onKey);
  }
  function onKey(e){
    if (e.key === 'Escape' || e.key === 'Enter' || e.key === ' ') {
      e.preventDefault();
      finish();
    }
  }

  if (prefersReduced || isNarrow || seen) {
    finish();
    return;
  }
  document.body.classList.add('intro-lock');
  ov.classList.add('is-leaving');
  ov.addEventListener('click', finish);
  if (skipBtn) skipBtn.addEventListener('click', function(e){ e.stopPropagation(); finish(); });
  document.addEventListener('keydown', onKey);
  setTimeout(finish, 3200);
})();
</script>
```

### Task 9: Light theme contrast audit (pairs that fail AA)

**Files:** `assets/tokens.css` and `assets/style-guild.css` only if a pair is below 4.5:1.

Measure on `/pathways.html` and `/quiz.html` with `html[data-theme="light"]`. Use this Python helper (bg/fg from computed styles or tokens):

```python
def srgb(c):
    c=c/255
    return c/12.92 if c<=0.04045 else ((c+0.055)/1.055)**2.4
def lum(hex):
    hex=hex.lstrip('#')
    r,g,b=[int(hex[i:i+2],16) for i in (0,2,4)]
    return 0.2126*srgb(r)+0.7152*srgb(g)+0.0722*srgb(b)
def contrast(a,b):
    L1,L2=sorted([lum(a),lum(b)], reverse=True)
    return (L1+0.05)/(L2+0.05)
```

Pairs to score (light theme):

| Use | Text token | Surface |
